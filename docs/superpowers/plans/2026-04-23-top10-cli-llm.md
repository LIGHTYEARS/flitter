# CLI + LLM — OpenAI Chat Completions, TOML Themes, IDE Flag Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add OpenAI chat.completions fallback (GAP-LLM-18), custom TOML theme loading (GAP-TUI-16), and `--ide`/`--no-ide` flag (GAP-CLI-32).

**Architecture:** OpenAI provider gains a `useChatCompletions` config flag that routes to `client.chat.completions.create` with field mapping and the existing compat transformer. TOML theme loader scans `~/.config/flitter/themes/*/colors.toml`, parses with a minimal TOML parser, and registers in `ThemeRegistry`. IDE flag is a boolean option that stores intent for future IDE client wiring.

**Tech Stack:** TypeScript, OpenAI SDK, TOML parsing, Commander.js.

```ts
// packages/tui/src/theme/toml-theme-loader.ts
// 逆向: c70/s70/o70/n70/A70 (chunk-004.js:30050) — theme discovery, loading, validation, mapping

import { readdirSync, readFileSync, statSync } from "fs";
import { join } from "path";
import { homedir } from "os";

interface RgbColor { r: number; g: number; b: number }

export interface ParsedTheme {
  name: string;
  mode: "dark" | "light";
  colors: {
    background: RgbColor;
    foreground: RgbColor;
    primary: RgbColor;
    success: RgbColor;
    warning: RgbColor;
    destructive: RgbColor;
    secondary?: RgbColor;
    accent?: RgbColor;
    info?: RgbColor;
  };
  ui?: {
    mutedForeground?: RgbColor;
    border?: RgbColor;
    cursor?: RgbColor;
    selection?: RgbColor;
    copyHighlight?: RgbColor;
    tableBorder?: RgbColor;
  };
  syntax?: Record<string, RgbColor>;
}

export function hexToRgb(hex: string): RgbColor | null {
  const match = hex.match(/^#([0-9a-f]{2})([0-9a-f]{2})([0-9a-f]{2})([0-9a-f]{2})?$/i);
  if (!match) return null;
  return { r: parseInt(match[1], 16), g: parseInt(match[2], 16), b: parseInt(match[3], 16) };
}

/** Minimal TOML parser — handles [sections], key = "value", and top-level strings. */
function parseTomlSimple(text: string): Record<string, Record<string, string> | string> {
  const result: Record<string, any> = {};
  let currentSection = "";

  for (const rawLine of text.split("\n")) {
    const line = rawLine.trim();
    if (!line || line.startsWith("#")) continue;

    const sectionMatch = line.match(/^\[([a-zA-Z_]+)\]$/);
    if (sectionMatch) {
      currentSection = sectionMatch[1];
      if (!result[currentSection]) result[currentSection] = {};
      continue;
    }

    const kvMatch = line.match(/^([a-zA-Z_]+)\s*=\s*"([^"]*)"$/);
    if (kvMatch) {
      if (currentSection) {
        result[currentSection] = result[currentSection] ?? {};
        result[currentSection][kvMatch[1]] = kvMatch[2];
      } else {
        result[kvMatch[1]] = kvMatch[2];
      }
    }
  }
  return result;
}

const REQUIRED_COLORS = ["background", "foreground", "primary", "success", "warning", "destructive"] as const;

export function parseThemeToml(tomlText: string, dirName: string): ParsedTheme | null {
  const parsed = parseTomlSimple(tomlText);
  const colorsSection = parsed.colors as Record<string, string> | undefined;
  if (!colorsSection || typeof colorsSection !== "object") return null;

  // Validate required fields
  for (const key of REQUIRED_COLORS) {
    if (!colorsSection[key]) return null;
  }

  const colors: any = {};
  for (const key of REQUIRED_COLORS) {
    const rgb = hexToRgb(colorsSection[key]);
    if (!rgb) return null;
    colors[key] = rgb;
  }

  // Optional color fields
  for (const key of ["secondary", "accent", "info"]) {
    if (colorsSection[key]) {
      const rgb = hexToRgb(colorsSection[key]);
      if (rgb) colors[key] = rgb;
    }
  }

  // Name: explicit or directory name
  const name = (typeof parsed.name === "string" ? parsed.name : dirName);

  // Mode: explicit or auto-detect from bg luminance
  let mode: "dark" | "light";
  if (typeof parsed.mode === "string" && (parsed.mode === "dark" || parsed.mode === "light")) {
    mode = parsed.mode;
  } else {
    const bg = colors.background;
    const luma = (0.299 * bg.r + 0.587 * bg.g + 0.114 * bg.b) / 255;
    mode = luma > 0.5 ? "light" : "dark";
  }

  // Optional UI section
  const uiSection = parsed.ui as Record<string, string> | undefined;
  let ui: ParsedTheme["ui"];
  if (uiSection && typeof uiSection === "object") {
    ui = {};
    for (const key of ["muted_foreground", "border", "cursor", "selection", "copy_highlight", "table_border"]) {
      if (uiSection[key]) {
        const rgb = hexToRgb(uiSection[key]);
        // Convert snake_case to camelCase
        const camelKey = key.replace(/_([a-z])/g, (_, c) => c.toUpperCase());
        if (rgb) (ui as any)[camelKey] = rgb;
      }
    }
  }

  // Optional syntax section
  const syntaxSection = parsed.syntax as Record<string, string> | undefined;
  let syntax: Record<string, RgbColor> | undefined;
  if (syntaxSection && typeof syntaxSection === "object") {
    syntax = {};
    for (const [key, val] of Object.entries(syntaxSection)) {
      const rgb = hexToRgb(val);
      if (rgb) syntax[key] = rgb;
    }
  }

  return { name, mode, colors, ui, syntax };
}

/** Scan theme directory for subdirectories containing colors.toml */
// 逆向: s70/c70 (chunk-004.js:30050)
export function scanThemeDirectory(themesDir?: string): ParsedTheme[] {
  const dir = themesDir ?? join(
    process.env.XDG_CONFIG_HOME ?? join(homedir(), ".config"),
    "flitter", "themes",
  );

  const themes: ParsedTheme[] = [];
  try {
    for (const entry of readdirSync(dir)) {
      const entryPath = join(dir, entry);
      if (!statSync(entryPath).isDirectory()) continue;

      const tomlPath = join(entryPath, "colors.toml");
      try {
        const content = readFileSync(tomlPath, "utf-8");
        const theme = parseThemeToml(content, entry);
        if (theme) themes.push(theme);
      } catch { /* no colors.toml or parse error — skip */ }
    }
  } catch { /* directory doesn't exist — no custom themes */ }

  return themes;
}
```
